fn imported_session_id(harness: HarnessKind, id: &str) -> String {
    let id = if harness == HarnessKind::Omp {
        id
    } else {
        id.char_indices()
            .nth(12)
            .map(|(byte_offset, _)| &id[..byte_offset])
            .unwrap_or(id)
    };
    format!("{}_{}", harness.as_str(), id)
}

/// Extracts plain text representation from Codex content field.
pub(crate) fn extract_codex_content_text(val: Option<&Value>) -> String {
    let Some(v) = val else {
        return String::new();
    };
    match v {
        Value::String(s) => s.clone(),
        Value::Array(arr) => arr
            .iter()
            .filter_map(|item| {
                item.get("text")
                    .or_else(|| item.get("summary_text"))
                    .and_then(|t| t.as_str())
            })
            .collect::<Vec<_>>()
            .join("\n"),
        other => serde_json::to_string(other).unwrap_or_default(),
    }
}

/// Imports a Claude Code / Codex Rollout / OpenCode / Pi JSONL session transcript file into `CwStore`.
pub fn import_generic_jsonl_transcript(
    path: &Path,
    harness: HarnessKind,
    store: &CwStore,
    workspace_root: Option<String>,
) -> Result<String, HarnessError> {
    let mut harness = harness;
    let path_str = path.to_string_lossy();
    if harness == HarnessKind::Pi && path_str.contains("/.omp/") {
        harness = HarnessKind::Omp;
    }
    let stem = path
        .file_stem()
        .and_then(|s| s.to_str())
        .unwrap_or("session");
    // For Codex `rollout-2026-10-19T09-46-31-01a15320-58da-...`, use the UUID suffix so sessions don't collide on `rollout-2026`
    let id_slug = if stem.starts_with("rollout-") && stem.len() > 28 {
        stem.get(28..).unwrap_or(stem)
    } else if let Some((_, uuid_part)) = stem.split_once('_') {
        uuid_part
    } else {
        stem
    };
    let mut session_id = imported_session_id(harness, id_slug);

    let file = fs::File::open(path)?;
    let file_len = file.metadata().map(|m| m.len()).unwrap_or(0);
    let lines_to_parse: Vec<String> = if file_len > 2_000_000 {
        use std::io::{Read, Seek, SeekFrom};
        let mut head_reader = BufReader::new(&file);
        let mut selected = Vec::with_capacity(600);
        for _ in 0..45 {
            let mut l = String::new();
            if head_reader.read_line(&mut l).unwrap_or(0) == 0 {
                break;
            }
            selected.push(l);
        }
        let mut tail_file = fs::File::open(path)?;
        let _ = tail_file.seek(SeekFrom::Start(file_len.saturating_sub(1_200_000)));
        let mut tail_buf = String::new();
        let _ = tail_file.read_to_string(&mut tail_buf);
        // Skip first partial line after seek
        for l in tail_buf.lines().skip(1) {
            selected.push(l.to_string());
        }
        selected
    } else {
        BufReader::new(file).lines().map_while(Result::ok).collect()
    };

    let mut messages = Vec::new();
    let mut system_chunks: Vec<String> = Vec::new();
    let mut turn_counter = 0u32;
    let mut detected_workspace = workspace_root;
    let mut detected_model = if harness == HarnessKind::Codex {
        "gpt-5-codex".to_string()
    } else {
        "claude-sonnet-4".to_string()
    };
    let mut detected_provider = if harness == HarnessKind::Codex {
        ProviderKind::OpenAI
    } else {
        ProviderKind::Anthropic
    };
    let mut title: Option<String> = None;
    let mut first_ts: Option<DateTime<Utc>> = None;
    let mut last_ts: Option<DateTime<Utc>> = None;
    let mut latest_input_tokens: Option<u32> = None;
    let mut latest_cache_read_tokens: Option<u32> = None;
    let mut latest_output_tokens: u32 = 0;

    for (line_idx, line) in lines_to_parse.into_iter().enumerate() {
        if line.trim().is_empty() {
            continue;
        }
        let Ok(entry) = serde_json::from_str::<Value>(&line) else {
            continue;
        };

        let ts_str = entry
            .get("timestamp")
            .or_else(|| entry.get("created_at"))
            .and_then(|v| v.as_str())
            .map(|s| s.to_string());
        if let Some(ref ts) = ts_str
            && let Ok(dt) = DateTime::parse_from_rfc3339(ts)
        {
            let utc_dt = dt.with_timezone(&Utc);
            if first_ts.is_none() {
                first_ts = Some(utc_dt);
            }
            last_ts = Some(utc_dt);
        }

        if detected_workspace.is_none()
            && let Some(cwd) = entry.get("cwd").and_then(|v| v.as_str())
            && !cwd.is_empty()
        {
            detected_workspace = Some(cwd.to_string());
        }

        let entry_type = entry.get("type").and_then(|v| v.as_str()).unwrap_or("");

        if entry_type == "title" {
            if let Some(t) = entry.get("title").and_then(|v| v.as_str())
                && !t.trim().is_empty()
            {
                title = Some(t.trim().to_string());
            }
            continue;
        }

        // 1. Codex CLI Rollout format (`session_meta`, `turn_context`, `event_msg`, `response_item`)
        if entry_type == "session" {
            if let Some(id) = entry.get("id").and_then(|v| v.as_str())
                && !id.is_empty()
            {
                session_id = imported_session_id(harness, id);
            }
            if let Some(cwd) = entry.get("cwd").and_then(|v| v.as_str())
                && !cwd.is_empty()
            {
                detected_workspace = Some(cwd.to_string());
            }
            continue;
        }

        if entry_type == "model_change" {
            if let Some(m) = entry.get("model").and_then(|v| v.as_str())
                && !m.is_empty()
            {
                detected_model = m.to_string();
            }
            continue;
        }

        if entry_type == "session_meta" || entry_type == "turn_context" {
            if let Some(p) = entry.get("payload") {
                if let Some(sid) = p.get("session_id").and_then(|v| v.as_str())
                    && !sid.is_empty()
                {
                    session_id = imported_session_id(harness, sid);
                }
                if let Some(cwd) = p.get("cwd").and_then(|v| v.as_str())
                    && !cwd.is_empty()
                {
                    detected_workspace = Some(cwd.to_string());
                }
                if let Some(m) = p.get("model").and_then(|v| v.as_str())
                    && !m.is_empty()
                {
                    detected_model = m.to_string();
                }
                if let Some(prov) = p.get("model_provider").and_then(|v| v.as_str()) {
                    if prov.contains("ollama") || prov.contains("local") {
                        detected_provider = ProviderKind::OpenAICompatible;
                    } else if prov.contains("openai") {
                        detected_provider = ProviderKind::OpenAI;
                    }
                }
            }
            continue;
        }

        if entry_type == "compacted" {
            let win_num = entry
                .get("payload")
                .and_then(|p| p.get("window_number"))
                .and_then(|w| w.as_u64())
                .unwrap_or(1);
            let rep_len = entry
                .get("payload")
                .and_then(|p| p.get("replacement_history"))
                .and_then(|r| r.as_array())
                .map(|a| a.len())
                .unwrap_or(0);
            messages.clear();
            turn_counter = turn_counter.max(win_num as u32) + 1;
            messages.push(json!({
                "role": "assistant",
                "content": [{
                    "type": "text",
                    "text": format!("[cw:folded_turns 1..{}] Codex Compacted Context Window #{} ({} replacement items retained)", turn_counter, win_num, rep_len)
                }],
                "cw_timestamp": ts_str,
                "cw_turn_seq": turn_counter
            }));
            continue;
        }

        if entry_type == "event_msg" {
            if let Some(p) = entry.get("payload")
                && p.get("type").and_then(|v| v.as_str()) == Some("token_count")
                && let Some(last_usage) = p.get("info").and_then(|i| i.get("last_token_usage"))
            {
                if let Some(inp) = last_usage.get("input_tokens").and_then(|v| v.as_u64()) {
                    latest_input_tokens = Some(inp as u32);
                }
                if let Some(cached) = last_usage
                    .get("cached_input_tokens")
                    .and_then(|v| v.as_u64())
                {
                    latest_cache_read_tokens = Some(cached as u32);
                }
                if let Some(out) = last_usage.get("output_tokens").and_then(|v| v.as_u64()) {
                    latest_output_tokens = out as u32;
                }
            }
            continue;
        }

        if entry_type == "response_item" {
            let Some(p) = entry.get("payload") else {
                continue;
            };
            let item_type = p.get("type").and_then(|v| v.as_str()).unwrap_or("");
            match item_type {
                "message" => {
                    let role = p.get("role").and_then(|v| v.as_str()).unwrap_or("user");
                    let text = extract_codex_content_text(p.get("content"));
                    if text.trim().is_empty() {
                        continue;
                    }
                    if role == "developer"
                        || text.starts_with("# AGENTS.md")
                        || text.starts_with("<INSTRUCTIONS>")
                    {
                        if system_chunks.len() < 4 {
                            system_chunks.push(text);
                        }
                    } else if role == "user" {
                        if !text.starts_with("<codex_internal_context") {
                            turn_counter += 1;
                            if title.is_none() {
                                let one_line =
                                    text.split_whitespace().collect::<Vec<_>>().join(" ");
                                title = Some(one_line.chars().take(78).collect());
                            }
                        } else if turn_counter == 0 {
                            turn_counter = 1;
                        }
                        messages.push(json!({
                            "role": "user",
                            "content": text,
                            "cw_timestamp": ts_str,
                            "cw_turn_seq": turn_counter.max(1)
                        }));
                    } else {
                        messages.push(json!({
                            "role": "assistant",
                            "content": [{"type": "text", "text": text}],
                            "cw_timestamp": ts_str,
                            "cw_turn_seq": turn_counter.max(1)
                        }));
                    }
                }
                "reasoning" => {
                    let summary_txt = extract_codex_content_text(p.get("summary"));
                    let thought = if !summary_txt.trim().is_empty() {
                        summary_txt
                    } else {
                        "[Codex encrypted reasoning trace]".to_string()
                    };
                    messages.push(json!({
                        "role": "assistant",
                        "content": [{"type": "thinking", "thinking": thought}],
                        "cw_timestamp": ts_str,
                        "cw_turn_seq": turn_counter.max(1)
                    }));
                }
                "function_call" | "custom_tool_call" => {
                    let call_id = p
                        .get("call_id")
                        .or_else(|| p.get("id"))
                        .and_then(|v| v.as_str())
                        .map(|s| s.to_string())
                        .unwrap_or_else(|| format!("codex_call_{line_idx}"));
                    let name = p.get("name").and_then(|v| v.as_str()).unwrap_or("exec");
                    let raw_args = p
                        .get("arguments")
                        .or_else(|| p.get("input"))
                        .cloned()
                        .unwrap_or(Value::Null);
                    let input_val = match raw_args {
                        Value::String(ref s) => serde_json::from_str::<Value>(s)
                            .unwrap_or_else(|_| json!({ "CommandLine": s })),
                        other => other,
                    };
                    messages.push(json!({
                        "role": "assistant",
                        "content": [{
                            "type": "tool_use",
                            "id": call_id,
                            "name": name,
                            "input": input_val
                        }],
                        "cw_timestamp": ts_str,
                        "cw_turn_seq": turn_counter.max(1)
                    }));
                }
                "function_call_output" | "custom_tool_call_output" => {
                    let call_id = p
                        .get("call_id")
                        .or_else(|| p.get("id"))
                        .and_then(|v| v.as_str())
                        .map(|s| s.to_string())
                        .unwrap_or_else(|| format!("codex_res_{line_idx}"));
                    let out_txt = extract_codex_content_text(p.get("output"));
                    messages.push(json!({
                        "role": "user",
                        "content": [{
                            "type": "tool_result",
                            "tool_use_id": call_id,
                            "content": out_txt
                        }],
                        "cw_timestamp": ts_str,
                        "cw_turn_seq": turn_counter.max(1)
                    }));
                }
                "agent_message" => {
                    let author = p
                        .get("author")
                        .and_then(|v| v.as_str())
                        .unwrap_or("subagent");
                    let text = extract_codex_content_text(p.get("content"));
                    if !text.trim().is_empty() {
                        messages.push(json!({
                            "role": "assistant",
                            "content": [{"type": "text", "text": format!("[{author}] {text}")}],
                            "cw_timestamp": ts_str,
                            "cw_turn_seq": turn_counter.max(1)
                        }));
                    }
                }
                _ => {}
            }
            continue;
        }

        // 2. Claude Code / OpenCode / Pi message format
        if let Some(msg) = entry.get("message") {
            if let Some(m) = msg.get("model").and_then(|v| v.as_str())
                && !m.is_empty()
            {
                detected_model = m.to_string();
            }
            if let Some(prov) = msg.get("provider").and_then(|v| v.as_str())
                && (prov.contains("machinist") || prov.contains("local") || prov.contains("ollama"))
            {
                detected_provider = ProviderKind::OpenAICompatible;
            }
            if let Some(api) = msg.get("api").and_then(|v| v.as_str())
                && api.contains("openai")
            {
                detected_provider = ProviderKind::OpenAICompatible;
            }
            if let Some(u) = msg.get("usage") {
                let cached = u
                    .get("cache_read_input_tokens")
                    .or_else(|| u.get("cacheRead"))
                    .and_then(|v| v.as_u64());
                if let Some(c) = cached {
                    latest_cache_read_tokens = Some(c as u32);
                }
                let total_prompt = msg
                    .get("contextSnapshot")
                    .and_then(|cs| cs.get("promptTokens"))
                    .and_then(|v| v.as_u64())
                    .or_else(|| {
                        let inp = u
                            .get("input_tokens")
                            .or_else(|| u.get("input"))
                            .and_then(|v| v.as_u64())?;
                        Some(inp + cached.unwrap_or(0))
                    });
                if let Some(inp) = total_prompt {
                    latest_input_tokens = Some(inp as u32);
                }
                if let Some(out) = u
                    .get("output_tokens")
                    .or_else(|| u.get("output"))
                    .and_then(|v| v.as_u64())
                {
                    latest_output_tokens = out as u32;
                }
            }
            let role_val = msg.get("role").and_then(|r| r.as_str()).unwrap_or("");
            if role_val == "user" || role_val == "assistant" {
                turn_counter += 1;
            }
            if role_val == "user" && title.is_none() {
                if let Some(c) = msg.get("content").and_then(|v| v.as_str())
                    && !c.trim().is_empty()
                {
                    let one_line = c.split_whitespace().collect::<Vec<_>>().join(" ");
                    title = Some(one_line.chars().take(78).collect());
                } else if let Some(arr) = msg.get("content").and_then(|v| v.as_array()) {
                    for item in arr {
                        if item.get("type").and_then(|t| t.as_str()) == Some("text")
                            && let Some(t) = item.get("text").and_then(|s| s.as_str())
                            && !t.trim().is_empty()
                        {
                            let one_line = t.split_whitespace().collect::<Vec<_>>().join(" ");
                            title = Some(one_line.chars().take(78).collect());
                            break;
                        }
                    }
                }
            }
            let mut m_clone = msg.clone();
            if let Some(obj) = m_clone.as_object_mut() {
                if let Some(ref ts) = ts_str {
                    obj.insert("cw_timestamp".to_string(), Value::String(ts.clone()));
                }
                obj.insert("cw_turn_seq".to_string(), json!(turn_counter.max(1)));
            }
            messages.push(m_clone);
        } else if entry.get("role").is_some() {
            if entry.get("role").and_then(|r| r.as_str()) == Some("user") {
                turn_counter += 1;
            }
            messages.push(entry);
        }
    }

    if messages.is_empty() {
        return Ok(session_id);
    }

    let created_at = first_ts.unwrap_or_else(Utc::now);
    let updated_at = last_ts.unwrap_or(created_at);
    let system_text = if system_chunks.is_empty() {
        format!("{} Agent Session", harness.as_str())
    } else {
        system_chunks.join("\n\n")
    };

    let payload = json!({
        "cw_analysis_incomplete": file_len > 2_000_000,
        "model": detected_model,
        "max_tokens": 8192,
        "system": system_text,
        "messages": messages
    });

    let session = Session {
        id: session_id.clone(),
        parent_session_id: None,
        harness,
        workspace_root: detected_workspace.clone(),
        model: detected_model,
        provider: detected_provider,
        title: title.or_else(|| Some(format!("{} ({id_slug})", harness.as_str()))),
        created_at,
        updated_at,
    };
    store.upsert_session(&session)?;
    if let Ok(abs_path) = fs::canonicalize(path) {
        let byte_len = fs::metadata(&abs_path).map(|m| m.len()).unwrap_or(0);
        let _ = store.set_session_source_path(&session_id, &abs_path.to_string_lossy(), byte_len);
    }

    let seq = turn_counter.max(1);
    let existing_mutations = store.list_mutations(&session_id).unwrap_or_default();
    let mut mutated_payload_opt = None;
    let mut saved_by_mutations = 0u32;
    if !existing_mutations.is_empty() {
        let mut cloned = payload.clone();
        let rep = apply_mutations_to_payload(
            &mut cloned,
            ProviderKind::Anthropic,
            &existing_mutations,
            seq,
        );
        saved_by_mutations = rep.saved_tokens;
        mutated_payload_opt = Some(cloned);
    }
    let payload_for_analysis = mutated_payload_opt.as_ref().unwrap_or(&payload);

    let turn_id = format!("turn_{}_{}", session_id, seq);
    let (mut turn, file_states) = analyze_turn_from_payload(
        turn_id,
        session_id.clone(),
        None,
        seq,
        payload_for_analysis,
        ProviderKind::Anthropic,
        None,
        latest_cache_read_tokens,
        detected_workspace.as_deref(),
    )?;
    turn.created_at = updated_at;
    turn.token_usage.output_tokens = latest_output_tokens;
    turn.token_usage.saved_by_mutations = saved_by_mutations;
    if let Some(real_inp) = latest_input_tokens
        && real_inp > turn.token_usage.input_tokens
    {
        let adjusted_inp = real_inp.saturating_sub(saved_by_mutations);
        turn.token_usage.input_tokens = adjusted_inp;
        if let Some(cached) = latest_cache_read_tokens {
            turn.token_usage.cached_tokens = cached.min(adjusted_inp);
            turn.kv_cache_report.hit_tokens = cached.min(adjusted_inp);
            turn.kv_cache_report.newly_cached_tokens = adjusted_inp.saturating_sub(cached);
            turn.kv_cache_report.hit_ratio = cached as f32 / adjusted_inp.max(1) as f32;
        }
    }
    store.save_turn(&turn, Some(&payload), mutated_payload_opt.as_ref(), None)?;
    store.save_file_states(&session_id, &file_states)?;
    check_and_apply_auto_heal(
        store,
        &session_id,
        &mut turn,
        &payload,
        seq,
        ProviderKind::Anthropic,
        detected_workspace.as_deref(),
    );
    Ok(session_id)
}

/// Readiness is a ladder; each rung requires evidence beyond the previous one.
/// Static inspection can reach at most `Configured`; `FunctionallyVerified` is only
/// granted via [`mark_functionally_verified`] after an observed proxied request.
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize)]
pub enum HarnessReadinessStatus {
    /// Binary not found on PATH.
    Unavailable,
    /// Binary found but `--version` did not run successfully.
    Detected,
    /// Binary found and executes (`--version` succeeded).
    Installed,
    /// Installed, and contextWatcher can generate an isolated proxy configuration for it.
    Configured,
    /// Configured, and a proxied request through contextWatcher was actually observed.
    FunctionallyVerified,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct HarnessReadinessReport {
    pub harness: HarnessKind,
    pub binary_name: String,
    pub binary_path: Option<PathBuf>,
    pub version: Option<String>,
    pub is_installed: bool,
    /// The adapter *declares* credential isolation (e.g. env filters). Declared, not verified.
    pub credential_isolation_declared: bool,
    /// contextWatcher can generate an isolated proxy-routing config for this harness.
    pub proxy_config_available: bool,
    pub functionally_verified: bool,
    pub verification_evidence: Option<String>,
    pub status: HarnessReadinessStatus,
    pub notes: Vec<String>,
}

/// Promote a `Configured` report to `FunctionallyVerified` given observed evidence
/// (e.g. a session id whose requests were recorded by the proxy). Refuses otherwise.
pub fn mark_functionally_verified(
    report: &mut HarnessReadinessReport,
    evidence: impl Into<String>,
) -> bool {
    if report.status != HarnessReadinessStatus::Configured {
        return false;
    }
    report.functionally_verified = true;
    report.verification_evidence = Some(evidence.into());
    report.status = HarnessReadinessStatus::FunctionallyVerified;
    true
}

/// Discovers a binary by name across standard PATH directories.
pub fn find_binary(binary_name: &str) -> Option<PathBuf> {
    if let Ok(path_var) = std::env::var("PATH") {
        for dir in std::env::split_paths(&path_var) {
            let candidate = dir.join(binary_name);
            if candidate.is_file() {
                // Check if executable (Unix check)
                #[cfg(unix)]
                {
                    use std::os::unix::fs::PermissionsExt;
                    if let Ok(meta) = candidate.metadata()
                        && meta.permissions().mode() & 0o111 != 0
                    {
                        return Some(candidate);
                    }
                }
                #[cfg(not(unix))]
                return Some(candidate);
            }
        }
    }
    None
}

/// Extracts version from a binary using `--version`.
pub fn extract_binary_version(binary_path: &Path) -> Option<String> {
    let output = Command::new(binary_path).arg("--version").output().ok()?;
    if output.status.success() {
        let text = String::from_utf8_lossy(&output.stdout);
        let trimmed = text.lines().next().unwrap_or("").trim().to_string();
        if !trimmed.is_empty() {
            return Some(trimmed);
        }
    }
    None
}

/// Evaluates harness readiness for controlled live execution.
pub fn check_harness_readiness(harness: HarnessKind) -> HarnessReadinessReport {
    let binary_name = match harness {
        HarnessKind::Codex => "codex",
        HarnessKind::Omp | HarnessKind::OpenCode => "omp",
        HarnessKind::Hermes => "hermes",
        HarnessKind::Aider => "aider",
        HarnessKind::ClaudeCode => "claude",
        HarnessKind::Pi => "pi",
        HarnessKind::Antigravity => "agy",
        HarnessKind::GenericProxy => "proxy",
    };

    let bin_path = find_binary(binary_name);
    let mut notes = Vec::new();
    let version = bin_path.as_deref().and_then(extract_binary_version);
    // Only adapters with a generated, isolated proxy config count as configurable.
    let (cred_declared, proxy_config) = match harness {
        HarnessKind::Codex => (true, true), // codex_proxy_config_args + env filter
        HarnessKind::Omp | HarnessKind::OpenCode => (false, true), // omp_proxy_models_config
        _ => (false, false),
    };

    let status = match (&bin_path, &version) {
        (None, _) => {
            notes.push(format!("Binary '{binary_name}' not found in PATH"));
            HarnessReadinessStatus::Unavailable
        }
        (Some(path), None) => {
            notes.push(format!("Found {} but `--version` failed", path.display()));
            HarnessReadinessStatus::Detected
        }
        (Some(path), Some(v)) => {
            notes.push(format!("Found {} ({v})", path.display()));
            if proxy_config {
                HarnessReadinessStatus::Configured
            } else {
                notes.push("No generated isolated proxy config for this harness".into());
                HarnessReadinessStatus::Installed
            }
        }
    };
    if status >= HarnessReadinessStatus::Configured {
        notes.push("Not functionally verified: no proxied request observed".into());
    }

    HarnessReadinessReport {
        harness,
        binary_name: binary_name.to_string(),
        is_installed: version.is_some(),
        binary_path: bin_path,
        version,
        credential_isolation_declared: cred_declared && status >= HarnessReadinessStatus::Installed,
        proxy_config_available: proxy_config,
        functionally_verified: false,
        verification_evidence: None,
        status,
        notes,
    }
}

/// Evaluates a suite of harnesses and returns an inventory.
pub fn audit_all_harness_readiness() -> Vec<HarnessReadinessReport> {
    vec![
        check_harness_readiness(HarnessKind::Codex),
        check_harness_readiness(HarnessKind::OpenCode),
        check_harness_readiness(HarnessKind::Hermes),
    ]
}

## 1. Mission & Core Philosophy

`contextWatcher` is a **local-first, ultra-low-latency intercepting proxy, analyzer, and active manipulation engine** for LLM agent harnesses (Antigravity/AGY, Codex CLI, Pi, OpenCode, Aider, Claude Code).

### Golden Architectural Invariants
1. **Zero-Drop Proxy Transparency:** In passthrough mode (no active mutations or breakpoints), the proxy must never corrupt provider-specific fields, SSE streaming chunks, thinking signatures (e.g., Anthropic `signature` in `thinking` blocks or Gemini `thoughtSignature`), or cache control headers.
2. **Canonical Context IR + Lossless Raw Envelope:** Every intercepted request or imported session transcript is parsed into a normalized **Canonical Context Graph (`ContextTurn` / `ContextNode`)** for analysis and UI rendering, while preserving a **lossless JSON pointer map** to the original raw payload so mutations only touch targeted JSON paths.
3. **Deterministic Mutation Overlay:** Harnesses hold their own history in memory and re-send the full unmutated conversation on every turn. Therefore, all proxy mutations (Prune, Fold, Shadow-Dedupe, Pin, Tool-Gate) are stored as **deterministic overlays keyed by content/structural hashes** (`NodeFingerprint`) and automatically re-applied on every subsequent request—and optionally synced back to the harness's disk transcript.
4. **KV-Cache Awareness First:** Every mutation must calculate its **KV Cache Impact** before execution. We distinguish between *Prefix-Mutating Operations* (which invalidate the KV cache after token index $k$) and *Tail-Only / Cache-Safe Operations* (which preserve the cached prefix).

## 4. Coding Conventions & Quality Gates

### Rust (`crates/*`)
* **Edition:** Rust 2024 (or latest stable 2021+ toolchain).
* **Error Handling:** Use `thiserror` in library crates (`cw-core`, `cw-store`, `cw-harnesses`) and `anyhow` in binary/application crates (`cw-cli`). Never `.unwrap()` on untrusted network payloads or transcript JSON lines.
* **Async & Streaming:** Use `tokio` and `axum`. Proxy streaming responses (`text/event-stream`) must stream chunks directly to the harness with $<2\text{ms}$ overhead while tapping chunks asynchronously to the state engine.
* **Testing:** Every mutation operator, KV cache prefix differ, and protocol normalizer in `cw-core` must have deterministic unit tests using recorded real-world fixtures (`tests/fixtures/{anthropic,openai,gemini,agy}/`).

### Web UI (`web/`)
* **Framework:** Svelte 5 (Runes: `$state`, `$derived`, `$effect`), SvelteKit (`adapter-static` for embedding into `cw-cli` via `rust-embed`).
* **Visualization:** `@xyflow/svelte` for the interactive Context DAG / Radix Tree; `d3-hierarchy` for the Token Treemap & KV Cache Flamegraph.
* **State Sync:** Single WebSocket connection (`/api/v1/ws`) driving reactive Svelte 5 stores.

### IntelliJ Plugin (`ide/cw-intellij/`)
* **Language:** Kotlin (JVM 17/21), IntelliJ Platform Gradle Plugin (2.x).
* **Design:** Lightweight client connecting to the local `cw` daemon (`http://127.0.0.1:4400/api/v1/ws`). No heavy tokenization inside the IDE process—delegate all analysis to the Rust daemon.

| Multimodal or unknown content shapes | Raw preservation; partial analysis | Disabled | Unknown records preserved | Unsupported |

The watcher follows registered JSONL sources only, filtered by the selected workspace. It reparses changed bounded snapshots on a blocking worker. Native SQLite live watching and validated incremental append cursors are not implemented.

`cw init` installs and enables a Hermes plugin that adds authenticated session headers in Hermes' `llm_request` middleware. Launch Hermes through `cw wrap hermes chat ...` so its built-in DeepSeek provider uses the local `/deepseek` proxy route. The plugin reads the private CW token file in-process, only adds it for that loopback route, and the proxy strips CW headers before forwarding. A two-turn isolated Hermes run against a local mock verified session reuse and committed pin reapplication. A bounded real Hermes/DeepSeek request through an isolated CW daemon returned the expected answer and recorded the provider session; it does not qualify signatures, latency, or other provider routes. Native Hermes SQLite write-back remains unsupported. The old `post_tool_call` proposal was insufficient because Hermes ignores its return value; see [Hermes middleware](https://github.com/NousResearch/hermes-agent/blob/main/website/docs/developer-guide/middleware.md).

`CW_SESSION_ID` is child-process metadata, not proxy authentication: the proxy binds an overlay only when a request carries both authenticated `x-cw-session-id` and `x-cw-control-token` headers. OMP v18.12.6, Claude Code 2.1.126, and Codex CLI 0.165.0 passed local mock checks for authenticated session association. A follow-up Claude Code tool probe showed its Bash child can read `ANTHROPIC_CUSTOM_HEADERS`, including the control token. Therefore `cw wrap` now strips `x-cw-*` headers and control-token environment variables from every non-Codex child; OMP and Claude wrapper requests are observation-only. Codex keeps its authenticated path because its temporary CLI configuration excludes `CW_CONTROL_TOKEN` from shell subprocesses; local mock checks verified session reuse and overlay reapplication.

OpenCode 1.17.3, Pi 1.0.4, and Aider 0.86.2 each sent authenticated session headers when configured through their native provider settings and bound the mock request to the configured session. The proxy stripped those headers before forwarding. These tests used throwaway values in temporary config files; they do not establish safety for untrusted tools running as the same OS user, and `cw wrap` does not create these configurations. Aider receives the documented `OPENAI_API_BASE` endpoint alias from `cw wrap`.

Gemini CLI 0.42.0 and Antigravity CLI 1.4.10 completed local mock calls through `cw wrap` with a dummy API key and local upstream. Antigravity required `modelProvider: "gemini"` in its temporary settings. With explicit `--session-id` values, the store recorded new `sess_…` observation sessions rather than either requested ID, so authenticated session binding is not supported by these wrapper paths. The installed Gemini CLI 0.42.0 source supports `GEMINI_CLI_CUSTOM_HEADERS` for provider requests, but its shell environment sanitizer explicitly preserves all `GEMINI_CLI_*` variables. This is fixture-level preservation evidence, not validation of a cryptographic provider signature. A live OpenRouter-to-Anthropic signature check passed as described below; direct Anthropic connectivity and Gemini signatures remain unverified.

The control token protects the local API from unauthenticated callers; it does not isolate against a hostile process running as the same OS user with access to the token file. Run untrusted harnesses under an OS sandbox or separate account.

## Mutation rules

A pointer locates a target but does not establish its identity. A fingerprint or real call ID must match; when both are supplied both must match. System-fragment writes are disabled without an exact span map. All file-read results and current unread results are protected from generic pruning/folding/truncation. Covered-read dedupe additionally requires matching successful contents and covering ranges. A later write alone does not establish coverage.

Automatic folds, automatic spills, automatic transcript writes, unused-tool removal and local-engine parameter injection are disabled. User-created manual folds require supplied summary text and remain potentially lossy. Pins can change a prefix and are not equivalent to provider-native instructions or native transcript exports.

## Measurements and engine features

Only explicit provider usage is observed cache evidence. Radix/node spans, salience, model-family memory, cost and prefill calculations remain estimates. Unknown cache state is not a cache hit. Minimum optimization (`cache_safe_boost` / `min`) selects only zero-KV-break candidates (`kv_break_tokens == 0`), leaving any `CachedHit` prefix untouched.

Engine slot listing uses actual engine responses; offline slot lists are empty. Splicing and page-alignment calculations are previews. Execute requests for physical page manipulation return unsupported. Replay is a non-network preview. Real task continuation is a separate explicit benchmark operation.

### 4.2 Hermes Agent (`0.22.5`) — Concurrency Hardened & Resume Verified
* **Status:** **Harness Binary Verified (Multi-Turn Resume & Concurrency Hardened)** · Completed in CW-113.
* **Wire Protocol:** OpenAI / DeepSeek Chat Completions (`/v1/chat/completions`) with delta chunk streaming.
* **Middleware Extension Point & Concurrency Hardening (CW-113):**
  - `plugins/hermes/context-watcher/__init__.py` registers an `llm_request` hook that attaches `x-cw-session-id`, `x-cw-control-token`, and `x-cw-workspace` to requests routed to `_CW_ROUTES`.
  - **UUID-Scoped Override Handoff:** Replaced the global `hermes-session-override.json` file with unique per-invocation files: `~/.local/share/context-watcher/hermes-override-<uuid>.json`.
  - **Atomic File Creation & Strict Permissions:** Files are created using `OpenOptions::new().create_new(true).mode(0o600)`, ensuring no symlink or pre-existing file takeover and restricting access strictly to the invoking user.
  - **Environment-Directed Reference:** The wrapper passes the unique path via `CW_HERMES_OVERRIDE_FILE`. Hermes gateway allows this through via `terminal.client_env_passthrough`.
  - **Handoff Validation & Atomic Unlink:** The middleware verifies that the target path is a regular file (not a symlink), owned with mode `0600` (`mode & 077 == 0`), created within 120 seconds, and contains valid JSON with a non-empty session ID. It unlinks the file immediately on read (`unlink()`), so each handoff file is consumed exactly once.
  - **Gateway Daemon Session Scoping:** The Hermes gateway service (`hermes-gateway`) runs as a single persistent daemon serving multiple concurrent client sessions. The middleware indexes consumed overrides strictly by `hermes_sid` in `_SESSION_OVERRIDES`, eliminating shared module-level state and cross-session contamination.

In an uncontained execution environment, processes running under the same user UID share filesystem read privileges; isolation across processes requires OS containers (e.g. bubblewrap/docker). ### 4.3 OMP / Oh My Pi (`18.12.6`) — Native Runtime Context Hook & Hardened Isolation
* **Status:** **Harness Binary Verified (Native Runtime Hardened & Multi-Turn Resume Verified)** · Completed in CW-111 / CW-111B.
* **Wire Protocol:** OpenAI Chat Completions (`/v1/chat/completions`) & Anthropic Messages (`/v1/messages`).
* **Architecture & Security Boundary (CW-111B Hardening):**
  - `plugins/pi-opencode/index.js` registers native OMP runtime event hooks `pi.on("context", ...)` and `pi.on("before_provider_request", ...)`.
  - **Zero Credential Exposure in Config:** `models.json` contains zero custom headers or privileged CW control tokens. The unsafe helper `omp_proxy_models_config_with_headers()` has been completely removed from `cw-harnesses`.
  - **Process Environment Sanitization & Same-User Boundary:** Private control tokens are read securely from `~/.local/share/context-watcher/control-token` (`0600` permissions). The extension immediately purges `CW_CONTROL_TOKEN` and `CW_CONTROL_TOKEN_FILE` from `process.env` in memory on startup so child processes spawned by OMP tools (bash, scripts) cannot inherit them.
    > [!NOTE] Same-User Security Boundary: Files with `0600` permissions protect credentials against unauthorized user accounts on the host. However, any uncontained tool running under the *same* OS user account can access user-readable files unless enclosed in an OS sandbox/container (e.g. bubblewrap/docker). ContextWatcher purges the token from the process environment so standard tool executions do not inherit it.
  - **Loopback Enforcement:** `CW_DAEMON_URL` strictly validates hostname against loopback addresses (`127.0.0.1`, `localhost`, `::1`, `[::1]`). Any non-loopback destination fails closed back to local default `http://127.0.0.1:4400`.
  - **Canonical Session Identity & Isolation:** Prioritizes native session identity (`ctx.sessionManager.getSessionId()`) over ambient environment variables (`CW_SESSION_ID`), guaranteeing that switching native sessions within one process does not cross-bind to an initial wrapper session.
  - **Single-Pass Transformation & Reference Idempotency:** Employs reference-based `WeakSet` tracking on both transformed message arrays and constituent message objects. Eliminates heuristic time/length windows, guaranteeing that subsequent requests bypassing the context hook are safely transformed while turns already transformed by the context hook are never double-mutated.
  - **Dynamic Provider Resolution:** Automatically resolves provider (`anthropic`, `gemini`, `deepseek`, `ollama`, `openai`) from `ctx.model.provider` or payload shape.
  - **Fail-Open Guarantee:** If the CW daemon is offline, unauthenticated, or returns an error, the extension fails open instantly, returning pristine messages so agent workflows are never interrupted.

## 8. Empirical Performance & Savings Qualification (CW-115)

Implemented in [`crates/cw-proxy/tests/performance_qualification.rs`](../crates/cw-proxy/tests/performance_qualification.rs) and documented in [`docs/CW-115-PERFORMANCE-SAVINGS-QUALIFICATION.md`](CW-115-PERFORMANCE-SAVINGS-QUALIFICATION.md), with raw machine-readable telemetry recorded in [`docs/cw115_benchmark_data.json`](cw115_benchmark_data.json).

* **Scope & Methodology:** Evaluates 3 distinct experiment modes across real coding-agent workloads with real provider telemetry, session-level budget admission control ($1.00 ceiling), and loop limits (`MAX_REQUESTS_PER_SESSION = 30`) in the **Codex CLI (`0.166.0`) F1 Bug-Fixing Live Pilot**:
  - **Mode A (Native Baseline):** Harness connects directly to provider without proxy or intermediate hops.
  - **Mode B (CW Passthrough):** Harness routes through `cw-proxy` with active mutations disabled (`AutoPilotConfig.enabled = false`, 0 mutations).
  - **Mode C (CW Optimized):** Harness routes through `cw-proxy` with active committed context optimizations (historical tool output pruning and tail pin overlays).
* **Observed Empirical Findings:**
  - **Passthrough Overhead (Mode B vs Mode A):** Mean streaming overhead $< 2\text{ms/chunk}$; byte-level forwarding fidelity on intercepted payloads; TTFB averaging ~628ms; wall time within normal provider endpoint variation (-17.8% net delta due to cloud inference cluster timing); token totals reflect stochastic agent tool iteration paths (-90,190 mean delta).
  - **Headline Aggregate Finding (Mode C vs Mode B, Both Repetitions):** Across both repetitions, optimization increased total input tokens by **+50.0%** (451,353 vs 300,976) and calculated cost by **+46.2%** ($0.03950 vs $0.02702) due to mid-session prompt cache invalidation in Repetition 1.
  - **Cache-Aligned Counterexample (Repetition 2):** When optimizations preserved the prompt cache prefix, Mode C reduced input tokens by **-21.3%** (144,528 $\to$ 113,725), wall clock execution time by **-29.2%** (25.89s $\to$ 18.34s), and calculated cost by **-20.7%** ($0.01300 $\to$ $0.01031).
  - **Phase 2 Four-Mode Optimization Isolation Experiment:** Isolating optimization into Mode 0 (Passthrough Control), Mode 1 (Cache-Safe Tail Pin), Mode 2 (Historical Pruning), and Mode 3 (Combined Optimization) across live Codex CLI `gpt-4o-mini` runs (documented in `docs/CW-115-FOUR-MODE-ISOLATION-REPORT.md` and `docs/cw115_four_mode_benchmark_data.json`):
    - **Mode 1 (Tail-Only PinContext):** Retained 92.24% cache hit ratio (vs Mode 0 89.57%) in the live run; bit-for-bit prefix preservation proven deterministically via wire fixture test; observed -34.06% input token reduction via behavioral model steering (17 vs 23 requests), -37.71% cost savings ($0.01992 vs $0.03197), and -39.75% wall time speedup (36.4s vs 60.4s).
    - **Mode 2 (Historical Pruning Only):** Cut 84,505 input tokens (-24.15%) and reduced cost by -25.49% ($0.02382 vs $0.03197) when the pruned stub remained consistent across continuation turns.
    - **Mode 3 (Combined Optimization):** Delivered compound maximum reduction of -50.02% input tokens (174,920 vs 349,980), -43.5% model requests (13 vs 23), and -48.31% cost savings ($0.01653 vs $0.03197).
  - **Architectural Validation:** Confirms ContextWatcher's core architectural principle: **KV-Cache Awareness First**. Pruning historical turns when a warm cache exists destroys the 50% prompt-cache discount and induces exploratory requests; mutations must operate strictly in tail/cache-safe mode (`min` strategy) whenever a warm prefix is detected.
* **Execution:**
